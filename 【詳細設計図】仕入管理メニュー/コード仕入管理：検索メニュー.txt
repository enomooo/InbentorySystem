sequenceDiagram
    actor User
    participant IndexComp as ShiireMenu
    participant Service as ShiireService
    participant Repo as ShiireRepository
    participant DB as DB(PostgreSQL)

    title 仕入管理：検索メニューと処理

    %% --------------------
    %% 画面アクセスと初期表示
    %% --------------------
    User->>IndexComp: 画面アクセス（URL: /shiire/menu）
    activate IndexComp

    IndexComp-->>User: 初期表示

    %% --------------------
    %% 仕入検索処理
    %% --------------------
    User->>IndexComp: 仕入れ年月を入力 (例: 2025/10)
    User->>IndexComp: 商品コードを入力 (例: P001, または空)
    User->>IndexComp: 仕入検索ボタン クリック

    IndexComp->>Service: SearchShiireAsync()
    activate Service

    Service->>Repo: SearchByMonthAsync()
    activate Repo

    Repo->>DB: SELECT * FROM T_SHIIRE WHERE shiire_bi LIKE '[年月]%' AND shohin_code = '[コード]'
    activate DB
    DB-->>Repo: 該当仕入データ（List<ShiireModel>）
    deactivate DB
    Repo-->>Service: 検索結果
    deactivate Repo

    Service-->>IndexComp: 検索結果
    deactivate Service

    %% --------------------
    %% 異常系 (データなし) の処理と中断 (break)
    %% --------------------
    break 該当仕入伝票がない場合
        IndexComp-->>User: "該当する仕入伝票はありませんでした"と**Menu画面に表示**
        IndexComp->>IndexComp: 処理を中断 (return)
    end

    %% --------------------
    %% 正常系 (データあり) の処理
    %% --------------------
    IndexComp->>IndexComp: **仕入一覧画面**へ遷移（URL: /shiire/select）
    IndexComp-->>User: 検索結果を表示（URL: /shiire/select）
    deactivate IndexComp
